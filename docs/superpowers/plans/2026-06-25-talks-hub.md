# Talks Hub Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publish self-contained HTML presentations from anywhere via a global `publish-talk` CLI to a searchable overview at `svbaelen.me/talks`.

**Architecture:** A single bash script (`scripts/publish-talk.sh`, symlinked onto PATH) operates on a cached clone of this repo: it copies a one-file HTML presentation into `talks/slides/<slug>.html`, upserts an entry in `talks/manifest.json` via `jq`, commits, and pushes. A dependency-free static page (`talks/index.html`) fetches the manifest and renders a client-side searchable card list. No build step — GitHub Pages serves `talks/` directly from `main`.

**Tech Stack:** Bash, `jq`, `git`; vanilla HTML/CSS/JS reusing the existing `styles.css`.

## Global Constraints

- Self-contained single HTML files only — one file per presentation, no asset folders.
- No backend; overview + search are static client-side only; zero JS dependencies.
- Site deploys via push to root of `main` (GitHub Pages legacy mode). No build step.
- Cache clone lives at `~/.cache/webpage-personal`. CLI symlinked to `~/.local/bin/publish-talk`.
- Required runtime deps: `git`, `jq`. Script must check and fail clearly if missing.
- Keep `textwidth` ~80 chars where reasonable. No "Generated with Claude" / co-author lines in commits.
- Manifest entry shape (verbatim): `slug` (string), `title` (string), `date` (YYYY-MM-DD), `tags` (array of strings), `description` (string), `project` (string), `type` (string), `file` (string, relative path `slides/<slug>.html`).

```bash
#!/usr/bin/env bash
set -euo pipefail

REMOTE="${TALKS_REMOTE:-git@github.com:svbaelen/webpage-personal.git}"
REPO_DIR="${TALKS_REPO_DIR:-$HOME/.cache/webpage-personal}"

slugify() {
  printf '%s' "$1" \
    | tr '[:upper:]' '[:lower:]' \
    | sed -E 's/[^a-z0-9]+/-/g; s/-+/-/g; s/^-//; s/-$//'
}

die() { printf 'publish-talk: %s\n' "$1" >&2; exit 1; }

# When sourced for unit tests, define functions and stop here.
if [ "${PUBLISH_TALK_LIB:-0}" = "1" ]; then
  return 0 2>/dev/null || true
fi

# ---- parse args ----------------------------------------------------------
TITLE="" SLUG="" TAGS="" PROJECT="" TYPE="" DATE="" DESC="" FILE=""
CMD="publish"
while [ $# -gt 0 ]; do
  case "$1" in
    --title)   TITLE="$2"; shift 2;;
    --slug)    SLUG="$2"; shift 2;;
    --tags)    TAGS="$2"; shift 2;;
    --project) PROJECT="$2"; shift 2;;
    --type)    TYPE="$2"; shift 2;;
    --date)    DATE="$2"; shift 2;;
    --desc)    DESC="$2"; shift 2;;
    --list)    CMD="list"; shift;;
    --remove)  CMD="remove"; SLUG="$2"; shift 2;;
    -h|--help) CMD="help"; shift;;
    -*)        die "unknown flag: $1";;
    *)         FILE="$1"; shift;;
  esac
done

echo "cmd=$CMD title=$TITLE slug=$SLUG file=$FILE"  # placeholder; later tasks replace
```

- [ ] **Step 3: Write minimal implementation**

```bash
ensure_repo() {
  if [ ! -d "$REPO_DIR/.git" ]; then
    git clone -q "$REMOTE" "$REPO_DIR" || die "clone failed: $REMOTE"
  else
    git -C "$REPO_DIR" pull -q --rebase || die "pull failed"
  fi
}

manifest_upsert() {  # args: slug title date tags project type desc file
  local m="$REPO_DIR/talks/manifest.json" tmp
  tmp="$(mktemp)"
  [ -f "$m" ] || echo '[]' > "$m"
  # tags: comma list -> json array (empty string -> [])
  jq \
    --arg slug "$1" --arg title "$2" --arg date "$3" \
    --arg tags "$4" --arg project "$5" --arg type "$6" \
    --arg desc "$7" --arg file "$8" '
    ($tags | if . == "" then [] else split(",") | map(gsub("^ +| +$";"")) end) as $t
    | ([{slug:$slug,title:$title,date:$date,tags:$t,description:$desc,
         project:$project,type:$type,file:$file}]
       + map(select(.slug != $slug)))
    | sort_by(.date) | reverse
  ' "$m" > "$tmp" && mv "$tmp" "$m"
}

if [ "$CMD" = "publish" ]; then
  ensure_repo
  [ -n "$SLUG" ] || SLUG="$(slugify "$TITLE")"
  [ -n "$SLUG" ] || die "could not derive slug from title"
  [ -n "$DATE" ] || DATE="$(date +%F)"
  [ -n "$TITLE" ] || TITLE="$SLUG"
  mkdir -p "$REPO_DIR/talks/slides"
  cp "$FILE" "$REPO_DIR/talks/slides/$SLUG.html"
  manifest_upsert "$SLUG" "$TITLE" "$DATE" "$TAGS" "$PROJECT" "$TYPE" \
    "$DESC" "slides/$SLUG.html"
  git -C "$REPO_DIR" add talks
  git -C "$REPO_DIR" commit -q -m "Publish talk: $SLUG"
  git -C "$REPO_DIR" push -q
  printf 'Published: https://svbaelen.me/talks/slides/%s.html\n' "$SLUG"
fi
```

```bash
git_push_retry() {
  if ! git -C "$REPO_DIR" push -q 2>/dev/null; then
    git -C "$REPO_DIR" pull -q --rebase || die "push failed; pull --rebase failed"
    git -C "$REPO_DIR" push -q || die "push failed after rebase"
  fi
}
```

```bash
if [ "$CMD" = "list" ]; then
```

```html
<!DOCTYPE html>
<html lang="en">
  <head>
    <title>Talks &amp; Slides — Senne Van Baelen</title>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <link rel="shortcut icon" href="data:image/x-icon;," type="image/x-icon">
    <link rel="stylesheet" href="../styles.css">
    <style>
      .search { width: 100%; padding: .6em .8em; font: inherit;
                margin: 1.2em 0; box-sizing: border-box; }
      .talk { padding: .8em 0; border-top: 1px solid currentColor; }
      .talk h2 { margin: 0 0 .2em; font-size: 1.05rem; }
      .talk .meta { opacity: .7; font-size: .85rem; }
      .talk .tags span { display: inline-block; margin-right: .4em;
                         opacity: .8; font-size: .8rem; }
      .empty { opacity: .6; }
    </style>
  </head>
  <body>
    <main id="about">
      <section class="about-item">
        <h1 id="title">Talks &amp; Slides</h1>
        <input id="search" class="search" type="search"
               placeholder="Search by title, tag, project…"
               autocomplete="off" aria-label="Search talks">
        <div id="list"></div>
      </section>
    </main>
    <script>
      const listEl = document.getElementById('list');
      const searchEl = document.getElementById('search');
      let talks = [];

      const esc = s => (s ?? '').replace(/[&<>"]/g, c => (
        {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));

      function haystack(t) {
        return [t.title, t.project, t.description, (t.tags||[]).join(' ')]
          .join(' ').toLowerCase();
      }

      function render(items) {
        if (!items.length) {
          listEl.innerHTML = '<p class="empty">No talks found.</p>';
          return;
        }
        listEl.innerHTML = items.map(t => `
          <article class="talk">
            <h2><a href="${esc(t.file)}">${esc(t.title)}</a></h2>
            <div class="meta">${esc(t.date)}${
              t.project ? ' · ' + esc(t.project) : ''}${
              t.type ? ' · ' + esc(t.type) : ''}</div>
            ${t.description ? `<p>${esc(t.description)}</p>` : ''}
            <div class="tags">${(t.tags||[]).map(
              x => `<span>#${esc(x)}</span>`).join('')}</div>
          </article>`).join('');
      }

      function apply() {
        const q = searchEl.value.trim().toLowerCase();
        render(!q ? talks : talks.filter(t => haystack(t).includes(q)));
      }

      fetch('manifest.json')
        .then(r => r.json())
        .then(data => { talks = data; apply(); })
        .catch(() => { listEl.innerHTML =
          '<p class="empty">Could not load talks.</p>'; });

      searchEl.addEventListener('input', apply);
    </script>
  </body>
</html>
```
